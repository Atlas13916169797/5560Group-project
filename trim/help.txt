aero5560_aero.p
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%
% Calculates the total aerodynamic and propulsive forces and moments on the
% aircraft including propulsion model for a propeller - IN BODY AXES.
%
% Usage:
% [ForceCoeff, MomentCoeff] = aero5560_aero(X, Xg, Xdot, U, FlightData)
%
% Input:
% X:            12x1 state vector [u;v;w;p;q;r;phi;theta;psi;Xe;Ye;Ze]
% Xg:           6x1 gust vector [u;v;w;p;q;r]
% Xdot:         12x1 state rate vector
% U:            5x1 control vector [dt;de;da;dr;df]
% FlightData:   Aircraft dynamics, derivative, inertial properties structure
%
% Output:
% ForceCoeff:   3x1 force coefficient [CX;CY;CZ]
% MomentCoeff:  3x1 moment coefficient [Cl;Cm;Cn]
%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
